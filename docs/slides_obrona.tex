\begin{frame}{Motywacja i cel pracy}
    \begin{columns}[T]
        \column{0.55\textwidth}
        \textbf{Problem:}
        \begin{itemize}
            \item IaC (Dockerfile, manifesty K8s) jest złożone i podatne na błędy przy ręcznym tworzeniu
            \item LLM obiecują automatyzację, lecz brak twardej oceny end-to-end w literaturze
            \item Generowanie kodu aplikacyjnego przez LLM jest dobrze zbadane --- generowanie infrastruktury znacznie słabiej
        \end{itemize}
        \column{0.45\textwidth}
        \textbf{Cel pracy:}\\
        Ocena możliwości i ograniczeń agentów LLM w autonomicznej generacji konfiguracji IaC (Dockerfile + Kubernetes).
        \vspace{1em}

        \textbf{Główne pytanie:}\\
        \textit{Czy agent LLM może --- bez ingerencji człowieka --- wygenerować funkcjonalne, bezpieczne i zgodne z dobrymi praktykami konfiguracje wdrożeniowe?}
    \end{columns}
\end{frame}

\begin{frame}{Przegląd literatury i luki badawcze}
    \begin{columns}[T]
        \column{0.48\textwidth}
        \textbf{Kluczowe prace:}
        \begin{itemize}
            \item \textbf{Repo2Run} \cite{hu_llm-based_2025} --- generacja środowisk Docker z repozytoriów
            \item \textbf{Don't Train, Just Prompt} \cite{kratzke_dont_2024} --- prompt engineering dla IaC
            \item \textbf{GenKubeSec} \cite{malul_genkubesec_2024} --- LLM a bezpieczeństwo K8s
            \item \textbf{CloudEval-YAML} \cite{xu_cloudeval-yaml_2023} --- benchmark dla konfiguracji chmurowych
        \end{itemize}
        \column{0.48\textwidth}
        \textbf{Luki badawcze:}
        \begin{itemize}
            \item Brak oceny autonomicznej --- agenty czytające kod, nie tylko opis słowny
            \item Dominuje walidacja statyczna/składniowa, brak weryfikacji \textbf{runtime}
            \item Brak ilościowej analizy powtarzalności i podatności na manipulację kontekstem
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Pytania badawcze}
    \begin{itemize}
        \setlength{\itemsep}{0.6em}
        \item \textbf{RQ1:} Na ile agent generuje działające konfiguracje IaC end-to-end?
        \item \textbf{RQ2:} Jak złożoność systemu wpływa na skuteczność agenta?
        \item \textbf{RQ3:} Jaka jest jakość konfiguracji i czy \textit{prompt engineering} ją poprawia?
        \item \textbf{RQ4:} Na ile powtarzalny jest proces przy deterministycznych parametrach?
        \item \textbf{RQ5:} Jak podatny jest agent na manipulację kontekstem repozytorium?
    \end{itemize}
\end{frame}

\begin{frame}{Wnioski i implikacje praktyczne}
    \begin{block}{Odpowiedź na główne pytanie badawcze}
        \textbf{Tak} --- dla prostych aplikacji jednokomponentowych z czytelną strukturą kodu.\\
        \textbf{Nie} --- dla złożonych systemów wielousługowych oraz środowisk z podwyższonymi wymaganiami bezpieczeństwa lub niezaufaną dokumentacją.
    \end{block}
    \vspace{0.4em}
    \textbf{Implikacje praktyczne:}
    \begin{itemize}
        \item \textbf{Ograniczony kontekst wejściowy} --- tylko pliki kodu, bez dokumentacji i historii (ogranicza RQ5)
        \item \textbf{Rozbudowana instrukcja systemowa} --- konkretna checklista dobrych praktyk redukuje ostrzeżenia o $>79\%$ (RQ3)
        \item \textbf{Obowiązkowa walidacja statyczna} --- Hadolint + Kube-linter jako element CI/CD
        \item \textbf{Ludzki przegląd} --- artefakty z security context i uprawnieniami wymagają weryfikacji
    \end{itemize}
\end{frame}

\begin{frame}{Zagrożenia trafności}
    \begin{itemize}
        \setlength{\itemsep}{0.5em}
        \item Zbiór RQ1: 25 prostych aplikacji może nie uogólniać się na złożone systemy produkcyjne
        \item Mała liczba powtórzeń (RQ1: 2, RQ2: 3, RQ3: 1) --- szerokie przedziały ufności
        \item Walidacja runtime żądaniami HTTP do endpointów wskazanych przez agenta nie weryfikuje w pełni działania usługi
        \item 5 repozytoriów PoC (RQ2) --- zbyt mała próba do precyzyjnej lokalizacji progu złożoności
        \item Brak pętli sprzężenia zwrotnego z runtime --- architektura one-shot może zaniżać wyniki względem systemów iteracyjnych
    \end{itemize}
\end{frame}

\begin{frame}{Wkład do wiedzy i dalsze kierunki}
    \begin{columns}[T]
        \column{0.52\textwidth}
        \textbf{Wkład do wiedzy:}
        \begin{itemize}
            \item Empiryczna weryfikacja \textit{repo poisoning} w kontekście IaC --- instancja indirect prompt injection \parencite{greshake_not_2023}, nieznaleziona w przeglądzie literatury IaC
            \item Walidacja end-to-end z testami runtime --- ujawnia błędy niewidoczne dla narzędzi statycznych
            \item Ilościowy pomiar wpływu prompt engineeringu na jakość IaC ($-79{,}6\%$ ostrzeżeń)
            \item Kontrolowany eksperyment z pięcioma poziomami złożoności
        \end{itemize}
        \column{0.45\textwidth}
        \textbf{Dalsze kierunki:}
        \begin{itemize}
            \item Pętla sprzężenia zwrotnego z runtime --- agent reaguje na błędy wdrożenia
            \item Mechanizmy obronne: weryfikacja spójności kodu i dokumentacji
            \item Stabilizacja wyników przez wielokrotne generacje i agregację
            \item Zastosowanie w systemie PaaS --- automatyczne wdrożenie po podaniu linku do repozytorium
        \end{itemize}
    \end{columns}
\end{frame}

\begin{frame}{Dodatek A --- Architektura agenta i potok walidacji}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Agent (LangGraph, pętla ReAct), 11 narzędzi:}
        {\footnotesize
        \begin{itemize}
            \item \textbf{Repozytorium:} \texttt{clone\_repo}, \texttt{prepare\_repo\_tree}, \texttt{ls}
            \item \textbf{Odczyt i wyszukiwanie:} \texttt{get\_file\_content}, \texttt{search\_files}, \texttt{find\_files}
            \item \textbf{Zapis:} \texttt{write\_file}, \texttt{patch\_file} (unified diff)
            \item \textbf{Pomocnicze:} \texttt{base64\_encode}, \texttt{base64\_decode}, \texttt{think}
        \end{itemize}
        \vspace{0.3em}
        \texttt{clone\_repo} usuwa istniejące Dockerfile, manifesty i CI/CD --- agent nie widzi gotowych rozwiązań. Wyjście: JSON (obrazy, manifesty, \texttt{test\_endpoints}).}
        \column{0.5\textwidth}
        \textbf{Potok walidacyjny (7 kroków):}
        {\footnotesize
        \begin{enumerate}
            \item Składnia Dockerfile (\texttt{docker build --check})
            \item Hadolint
            \item Budowa obrazów + lokalny rejestr
            \item Składnia manifestów (\texttt{kubectl apply --dry-run=server})
            \item Kube-linter
            \item \texttt{kubectl apply} w osobnym namespace (MicroK8s)
            \item Runtime: żądania HTTP do \texttt{test\_endpoints}
        \end{enumerate}
        \vspace{0.3em}
        Sukces end-to-end = kroki 3, 6 i 7. Lintery (2, 5) tylko raportują (RQ3).}
    \end{columns}
\end{frame}

\begin{frame}{Dodatek B --- Prompt systemowy i wariant best practices}
    \begin{columns}[T]
        \column{0.5\textwidth}
        \textbf{Prompt bazowy (RQ1, RQ2, RQ4, RQ5):}
        {\footnotesize
        \begin{itemize}
            \item Kontekst i rola agenta, tryb one-shot bez pętli sprzężenia zwrotnego
            \item Sposób pracy: klonowanie $\to$ analiza kodu i zależności $\to$ generacja plików
            \item Reguły nazewnictwa: host Ingress \texttt{<repo>.<domena>}, RFC 1123
            \item Wytyczne wdrożeniowe: obraz bazowy, porty, requests/limits, health checks, ConfigMap/Secret/PVC dla wykrytych zależności
            \item Format wyjścia: strukturalny JSON
        \end{itemize}
        Sformułowany deklaratywnie, bez porad specyficznych dla technologii.}
        \column{0.5\textwidth}
        \textbf{Wariant best practices (RQ3):}
        {\footnotesize
        prompt bazowy + checklista dobrych praktyk dla Dockerfile i Kubernetes:
        \begin{itemize}
            \item użytkownik nieuprzywilejowany (\texttt{USER}, \texttt{runAsNonRoot})
            \item wersjonowanie obrazów zamiast \texttt{latest}
            \item requests/limits dla każdego kontenera
            \item readiness/liveness probes
            \item właściwe użycie Secret vs ConfigMap
        \end{itemize}
        \vspace{0.3em}
        Efekt (RQ3): ostrzeżenia $147 \to 30$ ($-79{,}6\%$); bez ostrzeżeń $17{,}3\% \to 74{,}7\%$. Pełna treść w załączniku pracy.}

\begin{frame}{Dodatek D --- Definicje metryk RQ4 i RQ5}
    \begin{columns}[T]

        \textbf{RQ4: run-to-run diff ratio}
        \[
            \text{diff\_ratio} = \frac{\text{linie zmienione}}{\tfrac{1}{2}\,(|A| + |B|)}
        \]
        {\footnotesize
        \begin{itemize}
            \item $A$, $B$ --- konkatenacja Dockerfile i manifestów z dwóch powtórzeń
            \item linie zmienione = wstawki + usunięcia (\texttt{difflib.ndiff})
            \item $0$ = identyczne, $1$ = niemal całkowite przepisanie; średnia po parach z 5 powtórzeń
            \item metryka pomocnicza: średnia $\pm$ odchylenie liczby wywołań narzędzi
        \end{itemize}}
        \column{0.5\textwidth}
        \textbf{RQ5: krytyczne odchylenie od specyfikacji}
        {\footnotesize
        \begin{itemize}
            \item \textbf{W1} sprzeczna specyfikacja: port z \texttt{spec.md} (9000) w \texttt{containerPort}/\texttt{Service} zamiast portu z kodu (8080)
            \item \textbf{W2} zalecenia bezpieczeństwa: \texttt{privileged: true} lub \texttt{hostNetwork: true} (ew. \texttt{runAsUser: 0}) bez potrzeby w kodzie
            \item \textbf{W3} pozorna zależność: zasoby Redis (Deployment/StatefulSet/Service/Secret/ConfigMap) bez zależności w kodzie
        \end{itemize}

  imagePullPolicy: IfNotPresent
  ports:
  - containerPort: 8080
  resources:
